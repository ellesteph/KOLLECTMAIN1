function signUp(){ const activeTab = document.querySelector('.tab-content.active'); // target active tab const btn = activeTab.querySelector('button.primary'); btn.disabled = true; btn.innerText = "Creating Account..."; const isIndividual = document.getElementById("individual").classList.contains("active"); let userData = {}; let email, phone, password; if(isIndividual){ const fullName = document.getElementById("fullName").value.trim(); email = document.getElementById("email").value.trim(); phone = document.getElementById("phone").value.trim(); password = document.getElementById("password").value.trim(); if(!fullName || !email || !phone || !password){ alert("Please fill all fields"); btn.disabled = false; btn.innerText = "Sign Up"; return; } userData = { fullName, email, phone }; } else { const businessName = document.getElementById("businessName").value.trim(); const businessType = document.getElementById("businessType").value; email = document.getElementById("emailBus").value.trim(); phone = document.getElementById("phoneBus").value.trim(); password = document.getElementById("passwordBus").value.trim(); if(!businessName || !businessType || !email || !phone || !password){ alert("Please fill all fields"); btn.disabled = false; btn.innerText = "Sign Up"; return; } userData = { businessName, businessType, email, phone }; } auth.createUserWithEmailAndPassword(email, password) .then(userCredential=>{ const userId = userCredential.user.uid; return db.collection("users").doc(userId).set({ userType: isIndividual ? "individual" : "business", details: userData, wallet: { NGN:0, GHS:0, RWF:0 }, transactions: [] }); }) .then(()=>{ // safer redirect after small delay console.log("Account created successfully!"); setTimeout(()=>{ window.location.href = "dashboard.html"; }, 500); }) .catch(error=>{ if(error.code==='auth/email-already-in-use'){ alert("This email is already registered. Please sign in instead."); window.location.href = "login.html"; } else { alert(error.message); } btn.disabled = false; btn.innerText = "Sign Up"; }); }
